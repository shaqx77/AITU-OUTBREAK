cmake_minimum_required(VERSION 3.20)
project(AITU_OUTBREAK LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/glfw)
add_library(glad STATIC vendor/glad.c)
target_include_directories(glad PUBLIC vendor)
add_executable(AITU_OUTBREAK src/main.cpp)
target_include_directories(AITU_OUTBREAK PRIVATE vendor vendor/glm)
target_compile_definitions(AITU_OUTBREAK PRIVATE GLFW_INCLUDE_NONE NOMINMAX)
target_link_libraries(AITU_OUTBREAK PRIVATE glad glfw opengl32)
if(MSVC)
  target_compile_options(AITU_OUTBREAK PRIVATE /utf-8 /W3)
  target_link_options(AITU_OUTBREAK PRIVATE /PDBALTPATH:AITU_OUTBREAK.pdb)

endif()
add_custom_command(TARGET AITU_OUTBREAK POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:AITU_OUTBREAK>/assets)

enable_testing()
add_test(NAME gameplay_and_graphics COMMAND AITU_OUTBREAK --self-test)
